{{- /* Social / SEO meta. Link previews target WhatsApp + Telegram: og:image absolute https, JPEG/PNG, < 300KB. og:image = first local post image (static/ is mounted into assets), re-encoded as JPEG when too big or not JPEG/PNG; else /og-default.jpg (1200x630). */ -}} {{- $isPost := and .IsPage (eq .Section "posts") }} {{- $title := cond .IsHome site.Title .Title }} {{- $desc := "" }} {{- with .Description }}{{ $desc = . }}{{ else }}{{ $desc = .Plain }}{{ end }} {{- $desc = $desc | htmlUnescape | replaceRE `\s+` " " }}{{ $desc = trim $desc " " }} {{- if not $desc }}{{ $desc = site.Params.tagline }}{{ end }} {{- if gt (len $desc) 200 }}{{ $desc = printf "%s…" (strings.TrimRight " ,.;:" (substr $desc 0 196)) }}{{ end }} {{- $img := false }} {{- $own := false }} {{- $max := 300000 }} {{- /* first usable local image, in document order: an to the full-size upload comes before the thumbnail it wraps; images under 120px a side (smileys, thumbs, banners) are skipped */ -}} {{- range findRESubmatch `(?:src|href)="/((?:wp-content/upload|ext)/[^"?#]+\.(?i:jpe?g|png|gif))"` .Content 20 }} {{- if not $img }} {{- with resources.Get (index . 1) }} {{- if and (eq .ResourceType "image") (ge .Width 120) (ge .Height 120) }} {{- $r := . }} {{- $ok := and (in (slice "image/jpeg" "image/png") .MediaType.Type) (lt (len .Content) $max) (le .Width 1200) }} {{- if not $ok }} {{- $spec := cond (gt .Width 1200) "resize 1200x jpg q80" "jpg q80" }} {{- $r = .Process $spec }} {{- if ge (len $r.Content) $max }}{{ $r = .Process "resize 800x jpg q60" }}{{ end }} {{- end }} {{- if lt (len $r.Content) $max }}{{ $img = $r }}{{ $own = true }}{{ end }} {{- end }} {{- end }} {{- end }} {{- end }} {{- if not $img }}{{ $img = resources.Get "og-default.jpg" }}{{ end }} {{- if $isPost }} {{- range .GetTerms "authors" }} {{- end }} {{- range .GetTerms "tags" }} {{- end }} {{- end }}
